% ============================== 3. PROBLEM STATEMENT
\section{Problem Statement}
\begin{frame}{Problem Statement}
\begin{block}{Problem}
\small
How should \textbf{history} and \textbf{observation} compression be
\textbf{combined} in a frozen LLM agent, so that we keep the token savings of
both \textbf{without losing task success}?
\end{block}
\vspace{-0.1cm}
\begin{columns}[T]
\begin{column}{0.47\textwidth}
\begin{alertblock}{Why it matters}
\footnotesize
\begin{itemize}
  \item Context grows every step $\Rightarrow$ memory cost, distraction
  \item ACON's combined setting: \textbf{$-$11.4} points, worse than either stage
\end{itemize}
\end{alertblock}
\begin{block}{Requirements}
\footnotesize
\begin{itemize}
  \item Agent weights stay \textbf{frozen}
  \item Same \textbf{compression schedule} across settings
  \item Measure compression quality \textbf{directly}
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.51\textwidth}
\begin{block}{Formulation}
\footnotesize
Agent step:\; $a_t = M(h'_{t-1},\,o'_t;\,\theta)$\\[2pt]
Pipeline:\; $\pi=(f_o,\,f_h,\,x,\,\mathcal{K})$ --- compressors, input order,
trigger steps
\[
\max_{\pi}\;
\mathbb{E}[R(\pi)]-\lambda\,\mathbb{E}[C(\pi)]-\mu\,\mathbb{E}[L(\pi)]
\]
s.t.\ \ $\mathbb{E}[R(\pi)]\ge\mathbb{E}[R(\pi_\varnothing)]-\varepsilon,\;\;C(\pi_{ho})\le\min\{C(\pi_h),C(\pi_o)\}$\par\smallskip
\centering $R$: task reward \quad $C$: token cost \quad $L$: latency
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Literature Review}
\begin{frame}{Literature Review (1/2): Existing Methods}
\begin{columns}[T]
\begin{column}{0.32\textwidth}
\begin{block}{ACON \cite{ACON}}
\footnotesize
\begin{itemize}
  \item LLM summarises history and observations past a token threshold
  \item Guideline optimised in natural language
  \item \textbf{Peak tokens $\downarrow$ 26--54\%}, accuracy kept

\end{column}
\begin{column}{0.32\textwidth}
\begin{block}{SAC \cite{SAC}}
\footnotesize
\begin{itemize}
  \item Anchor tokens absorb context via bidirectional attention
  \item Only anchor KV is kept
  \item \textbf{54.95 F1} at $15\times$ (EPL 51.52)
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.32\textwidth}
\begin{block}{LLMLingua \cite{LLMLingua}}
\footnotesize
\begin{itemize}
  \item Drops low-information tokens using a small LM
  \item Task-agnostic, single-step
  \item Only \textbf{39.3\%} on AppWorld history
\end{itemize}
\end{block}
\end{column}
\end{columns}
\vspace{0.05cm}
\begin{columns}[T]
\begin{column}{0.32\textwidth}
\begin{block}{SWE-Pruner \cite{SWEPruner}}
\footnotesize
\begin{itemize}
  \item 0.6B line classifier guided by an agent-written goal hint
  \item \textbf{Tokens $\downarrow$ 23--38\%}
  \item Resolve rate $+$1.2--1.4 points
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.32\textwidth}
\begin{block}{MEM1 / MemAgent \cite{MEM1,MemAgent}}
\footnotesize
\begin{itemize}

\end{block}
\end{column}

\textbf{Open:} how stages \emph{compose}, and whether critical facts survive.
\end{alertblock}

% ============================== THEORY: ACON AND SAC
\section{ACON and SAC}
\begin{frame}{ACON --- Agent Context Optimization}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\begin{block}{Threshold-triggered compression}
\scriptsize
\[
h'_t=\begin{cases} f(h_t;\phi,\mathcal{P}_{\mathrm{hist}}) & |h_t|>T_{\mathrm{hist}}\\ h_t & \text{else}\end{cases}
\]
\[
o'_t=\begin{cases} f(o_t,h_{t-1};\phi,\mathcal{P}_{\mathrm{obs}}) & |o_t|>T_{\mathrm{obs}}\\ o_t & \text{else}\end{cases}
\]
\end{block}
\begin{block}{Objective (agent $\theta$ frozen)}
\scriptsize
\[
\max_{\psi=(\phi,\mathcal{P})}\ \mathbb{E}\big[R(s_T(\psi))\big]-\lambda\,\mathbb{E}\big[C(H'(\psi))\big]
\]
\end{block}
\end{column}
\begin{column}{0.52\textwidth}
\centering
\begin{tikzpicture}[
  font=\scriptsize,
  bx/.style={draw, rounded corners=2pt, thick, align=center,

\node[bx, fill=envblue!8, draw=envblue, right=5mm of ok, yshift=-5.8mm] (opt) {optimizer LLM\\contrasts $H$, $H'$};
\node[bx, fill=metviolet!8, draw=metviolet, below=6mm of opt] (p) {refined guideline\\$\mathcal{P}^{(k)}\!\to\!\mathcal{P}^{(k+1)}$};

\end{tikzpicture}
\begin{block}{Guideline optimisation and distillation}
\scriptsize
\begin{itemize}
  \item \textbf{UT:} learn from tasks that \emph{fail} after compression
        $\Rightarrow$ accuracy $\uparrow$
  \item \textbf{CO:} learn from tasks that \emph{succeed} $\Rightarrow$
        tokens $\downarrow$; under \$2 per benchmark
  \item \textbf{Distil} gpt-4.1\,$+\,\mathcal{P}^*$ into Qwen3-14B:
        99.1\% cost cut, $>$95\% accuracy kept
\end{itemize}
\end{block}
\end{column}
\end{columns}

\vspace{0.12cm}
{\scriptsize
\textbf{Paper:} M.~Kang \emph{et al.}, ``ACON: Optimizing Context Compression
for Long-horizon LLM Agents,'' \emph{ICML} 2026~\cite{ACON}. Model-agnostic
(no agent weight updates); evaluated on AppWorld~\cite{AppWorld}, OfficeBench
and 8-objective QA; code released by Microsoft. \textbf{This is our base
paper.}}
\end{frame}

\begin{center}

  {\textbf{Compressor}\\bidirectional attn\\+ anchor embedding};

\vspace{0.1cm}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Mechanism}
\scriptsize
\begin{itemize}
  \item Some context tokens are chosen as \textbf{anchors} and marked with a
        learned anchor embedding
  \item \textbf{Bidirectional attention} lets each anchor absorb information
        from the whole context
  \item Only the anchors' key--value pairs are passed on: compression ratio
        $r$ keeps $n/r$ of them
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{Training and results}
\scriptsize
\begin{itemize}
  \item \textbf{Autoencoding-free:} no reconstruction objective; trainable
        adapter pretrained on SlimPajama, fine-tuned on MRQA
  \item Llama-3.2-1B, $15\times$: \textbf{54.95} ID / 39.26 OOD F1
        (EPL 51.52)
  \item Output is a \textbf{KV cache}, not text --- needs model internals
\end{itemize}
\end{block}
\end{column}
\end{columns}

\vspace{0.12cm}
{\scriptsize
\textbf{Paper:} X.~Liu \emph{et al.}, ``Autoencoding-Free Context Compression
for LLMs via Contextual Semantic Anchors,'' \emph{ICLR} 2026~\cite{SAC}.
Evaluated on MRQA (6 in-domain, 6 out-of-domain sets) at $5\times/15\times/51\times$;
checkpoints and data caches released, which is what let us reproduce it.}
\end{frame}

\section{Proposed Work}
\begin{frame}{Proposed Work}
\begin{center}
\begin{tikzpicture}[
  font=\scriptsize,
  bx/.style={draw, rounded corners=2pt, very thick, align=center,
    text width=2.35cm, minimum height=0.85cm, inner sep=2pt},
  arr/.style={-{Latex[length=2mm]}, line width=0.8pt},
  env/.style={bx, fill=envblue!8,  draw=envblue},
  cmp/.style={bx, fill=cmpgreen!8, draw=cmpgreen},

  met/.style={bx, fill=metviolet!8, draw=metviolet}]
\node[env] (o) {\textbf{Environment}\\raw observation $o_t$};
\node[cmp, right=6mm of o]  (oc) {\textbf{Obs.\ compressor}\\$o'_t$ for this step};
\node[env, right=6mm of oc] (ag) {\textbf{Agent}\\acts on $h'_{t-1},o'_t$};
\node[ctl, below=7mm of o]  (buf){\textbf{Raw buffer}\\keeps $o_1,\dots,o_t$};
\node[cmp, right=6mm of buf](hc) {\textbf{History compr.}\\$h'_t$ from \emph{raw} $o$};
\node[ctl, right=6mm of hc] (sch){\textbf{Fixed schedule}\\replayed triggers};
\node[met, right=6mm of sch](cfr){\textbf{CFR probe}\\facts in $o$ vs.\ $h'_t$};
\draw[arr] (o)--(oc);
\draw[arr] (oc)--(ag);
\draw[arr] (o)--(buf);
\draw[arr] (buf)--(hc);
\draw[arr] (sch)--(hc);
\draw[arr] (hc.north) -- ++(0,0.25) -| (ag.south);
\draw[arr, dashed, metviolet] (hc.south) -- ++(0,-0.25) -| (cfr.south);
\end{tikzpicture}
\end{center}
\vspace{0.1cm}
\begin{columns}[T]
\begin{column}{0.325\textwidth}
\begin{block}{1. Decoupled order}
\scriptsize
$f_h$ reads the \emph{raw} observations from a side buffer
($x^{\mathrm{dec}}$), not the pruned ones. If $f_o$ only deletes content,
$V(o'_t)\subseteq V(o_t)$, so facts dropped by $f_o$ can still reach the
summary.
\end{block}
\end{column}
\begin{column}{0.325\textwidth}
\begin{block}{2. Fixed schedule}
\scriptsize
Replay the history-only triggers, $\mathcal{K}(\pi_{ho}):=\mathcal{K}^{h}$,
instead of recomputing them on pruned text. Then $\pi_h$ and $\pi_{ho}$ differ
only in $f_o$, so $I$ measures composition alone.
\end{block}
\end{column}
\begin{column}{0.325\textwidth}
\begin{block}{3. CFR metric}
\scriptsize
\vspace{-0.2cm}
\[
\mathrm{CFR}=\frac{\sum_{t\in\mathcal{K}}\sum_{v\in F_t}\mathbf{1}[v\sqsubseteq h'_t]}
{\sum_{t\in\mathcal{K}}|F_t|}
\]
\vspace{-0.25cm}

Verbatim survival of task-critical IDs, tokens, paths and numbers; computed
offline, no extra rollouts.
\end{block}
\end{column}
\end{columns}
\end{frame}

% ============================== 12. RESEARCH GAPS
\section{Research Gaps}
\begin{frame}{Research Gaps}
\begin{alertblock}{(a) Token savings $\neq$ time savings}
\footnotesize
\begin{itemize}
  \item Compressor is itself an LLM call, and rewriting history \textbf{resets the KV cache}
  \item ACON's median time per task: \textbf{73\,s} (none) $\to$ \textbf{88\,s} (history) $\to$ \textbf{102\,s} (observation)
\end{itemize}
\end{alertblock}
\vspace{0.15cm}
\begin{alertblock}{(b) Combined setting is schedule-confounded}
\footnotesize
\begin{itemize}
  \item Compressing observations changes \emph{when} history compression fires
  \item ACON's combined drop ($I=\mathbf{-9.5}$ points) mixes composition with the shifted schedule
\end{itemize}
\end{alertblock}
\vspace{0.15cm}
\begin{alertblock}{(c) Compression quality is measured only indirectly}
\footnotesize
\begin{itemize}
  \item Judged only by task success or F1 --- never whether \textbf{critical facts} survive
  \item A failure cannot be traced to information loss vs.\ agent error
\end{itemize}
\end{alertblock}
\end{frame}

\section{Future Work}
\begin{frame}{Future Work}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Build the proposed pipeline}
\footnotesize

        with the schedule held fixed
\end{enumerate}
\end{block}
\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{Experiments}
\footnotesize
\begin{itemize}
